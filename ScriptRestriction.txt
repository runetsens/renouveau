# Configuration des variables
$OU_Users = "OU=Communication et Relations publiques,OU=UTILISATEUR,DC=billu,DC=remindme,DC=lan"
$Group_Bypass = "bypass" # Nom du groupe d'exception

# Importer le module Active Directory
Import-Module ActiveDirectory

# Fonction pour générer les horaires
function Generate-LogonHours {
    # Générer un tableau de 21 octets pour LogonHours
    $hours = @(
        127, # Lundi : 7h30 à 20h
        127, # Mardi : 7h30 à 20h
        127, # Mercredi : 7h30 à 20h
        127, # Jeudi : 7h30 à 20h
        127, # Vendredi : 7h30 à 20h
        127, # Samedi : 7h30 à 20h
        0    # Dimanche : Pas de connexion
    )
    
    # Étendre chaque jour à 3 octets (21 octets au total)
    $binaryArray = @()
    foreach ($day in $hours) {
        $binaryArray += $day, 0, 0
    }
    return $binaryArray
}

# Générer les plages horaires
$PlageHoraire = Generate-LogonHours

# Récupérer les utilisateurs standards
$Users = Get-ADUser -Filter * -SearchBase $OU_Users -Properties MemberOf, LogonHours

foreach ($User in $Users) {
    # Vérifie si l'utilisateur est dans le groupe "bypass" ou un administrateur
    $UserGroups = Get-ADUser -Identity $User.SamAccountName -Properties MemberOf
    if ($UserGroups -contains $Group_Bypass -or $User.AdminCount -eq 1) {
        Write-Host "Utilisateur $($User.SamAccountName) exclu des restrictions." -ForegroundColor Green
        continue
    }

    # Vérifier si les horaires sont déjà appliqués
    if ($User.LogonHours -eq $PlageHoraire) {
        Write-Host "Les restrictions sont déjà appliquées pour $($User.SamAccountName)." -ForegroundColor Cyan
        continue
    }

    # Appliquer les restrictions horaires
    try {
        Set-ADUser -Identity $User.SamAccountName -Replace @{LogonHours = $PlageHoraire}
        Write-Host "Restrictions appliquées pour $($User.SamAccountName)." -ForegroundColor Yellow
    } catch {
        Write-Host "Erreur lors de l'application des restrictions pour $($User.SamAccountName) : $_" -ForegroundColor Red
    }
}

Write-Host "Configuration terminée !" -ForegroundColor Cyan
